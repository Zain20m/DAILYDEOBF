--[[
  ENVLOGGER - powerful Roblox/Luau environment behavior logger
  Repo: https://github.com/Zain20m/envlogger

  Runs a target script inside a fully instrumented sandbox and emits a
  structured behavior log: services, instances, HTTP, loadstring, remotes,
  task/spawn, filesystem, getgenv writes, and unknown global probes.

  Designed for Discord bots and CLI pipelines. Offline-safe (no real network).
]]

local CFG = rawget(_G, "__ENVLOGGER_CFG") or {}
local MARK = "\0EL "
local START = os.clock()
local MAX_STMTS = tonumber(CFG.max_stmts) or 8000
local MAX_DEPTH = tonumber(CFG.max_depth) or 24
local TIME_BUDGET = tonumber(CFG.time_budget) or 25
local CAPTURE_LOADSTRING = CFG.capture_loadstring ~= false
local SOFT_ERRORS = CFG.soft_errors ~= false

local stmtCount = 0
local lines = {}
local urls = {}
local remotes = {}
local loadstrings = {}
local genvWrites = {}
local unknownGlobals = {}
local services = {}
local errors = {}

local function now()
	return string.format("%.3f", os.clock() - START)
end

local function trunc(s, n)
	s = tostring(s)
	n = n or 160
	if #s > n then
		return s:sub(1, n) .. "..."
	end
	return s
end

local function fmtval(v, depth)
	depth = depth or 0
	local t = type(v)
	if t == "nil" then
		return "nil"
	elseif t == "boolean" or t == "number" then
		return tostring(v)
	elseif t == "string" then
		return string.format("%q", trunc(v, 120))
	elseif t == "function" then
		return "<function>"
	elseif t == "thread" then
		return "<thread:" .. coroutine.status(v) .. ">"
	elseif t == "userdata" then
		return "<userdata>"
	elseif t == "table" then
		if depth > 2 then
			return "{...}"
		end
		local n = 0
		local parts = {}
		for k, val in pairs(v) do
			n += 1
			if n > 6 then
				parts[#parts + 1] = "..."
				break
			end
			parts[#parts + 1] = "[" .. fmtval(k, depth + 1) .. "]=" .. fmtval(val, depth + 1)
		end
		return "{" .. table.concat(parts, ", ") .. "}"
	end
	return "<" .. t .. ">"
end

local function emit(kind, msg)
	if stmtCount >= MAX_STMTS then
		return
	end
	stmtCount += 1
	lines[#lines + 1] = string.format("[%s] #%d %s %s", now(), stmtCount, kind, msg)
end

local function noteError(where, err)
	local e = trunc(tostring(err), 240)
	errors[#errors + 1] = where .. ": " .. e
	emit("ERROR", where .. " -> " .. e)
end

-- ---------------------------------------------------------------------------
-- Proxy Instance model
-- ---------------------------------------------------------------------------
local InstanceMT = {}
InstanceMT.__index = InstanceMT

local function newInstance(className, parent)
	local self = setmetatable({
		ClassName = className or "Instance",
		Name = className or "Instance",
		Parent = parent,
		_children = {},
		_attrs = {},
	}, InstanceMT)
	if parent and type(parent) == "table" and parent._children then
		parent._children[#parent._children + 1] = self
	end
	return self
end

function InstanceMT:GetChildren()
	emit("INSTANCE", self.ClassName .. ":GetChildren() n=" .. tostring(#self._children))
	return self._children
end

function InstanceMT:FindFirstChild(name)
	emit("INSTANCE", self.ClassName .. ":FindFirstChild(" .. fmtval(name) .. ")")
	for _, c in ipairs(self._children) do
		if c.Name == name then
			return c
		end
	end
	return nil
end
