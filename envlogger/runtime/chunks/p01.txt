function InstanceMT:WaitForChild(name, timeout)
	emit("INSTANCE", self.ClassName .. ":WaitForChild(" .. fmtval(name) .. ")")
	local c = self:FindFirstChild(name)
	if c then
		return c
	end
	local child = newInstance("Folder", self)
	child.Name = tostring(name)
	return child
end

function InstanceMT:GetService(name)
	emit("SERVICE", "GetService(" .. fmtval(name) .. ")")
	services[tostring(name)] = (services[tostring(name)] or 0) + 1
	if not self._services then
		self._services = {}
	end
	if not self._services[name] then
		self._services[name] = newInstance(tostring(name), self)
		self._services[name].Name = tostring(name)
	end
	return self._services[name]
end

function InstanceMT:Destroy()
	emit("INSTANCE", self.ClassName .. ":Destroy()")
	self.Parent = nil
end

function InstanceMT:Clone()
	emit("INSTANCE", self.ClassName .. ":Clone()")
	return newInstance(self.ClassName, nil)
end

function InstanceMT:IsA(className)
	return self.ClassName == className or className == "Instance"
end

function InstanceMT:GetFullName()
	return self.Name
end

function InstanceMT:SetAttribute(k, v)
	emit("INSTANCE", self.ClassName .. ":SetAttribute(" .. fmtval(k) .. ", " .. fmtval(v) .. ")")
	self._attrs[k] = v
end

function InstanceMT:GetAttribute(k)
	emit("INSTANCE", self.ClassName .. ":GetAttribute(" .. fmtval(k) .. ")")
	return self._attrs[k]
end

function InstanceMT:GetDescendants()
	emit("INSTANCE", self.ClassName .. ":GetDescendants()")
	return {}
end

local function newSignal(name, owner)
	local sig = {
		_name = name,
		_owner = owner,
		Connect = function(self, fn)
			emit("SIGNAL", tostring(owner and owner.ClassName or "?") .. "." .. name .. ":Connect(<fn>)")
			return {
				Disconnect = function()
					emit("SIGNAL", name .. ":Disconnect()")
				end,
			}
		end,
		Once = function(self, fn)
			emit("SIGNAL", tostring(owner and owner.ClassName or "?") .. "." .. name .. ":Once(<fn>)")
			return {
				Disconnect = function() end,
			}
		end,
		Wait = function()
			emit("SIGNAL", name .. ":Wait()")
			return nil
		end,
		Fire = function(self, ...)
			emit("SIGNAL", name .. ":Fire(" .. trunc(table.concat({
				fmtval(select(1, ...)),
			}, ", "), 100) .. ")")
		end,
	}
	return sig
end
